2. Creating Custom Logger Middleware
You can also write a simple, custom logger function to understand how middleware works. 
Usage:
javascript
const express = require('express');
const app = express();

// Custom logger middleware function
const myLogger = function (req, res, next) {
  console.log(`LOGGED: ${req.method} request to ${req.url}`);
  next(); // Pass control to the next middleware or route handler
};

// Load the custom middleware
app.use(myLogger);

app.get('/', (req, res) => {
  res.send('Hello from custom logger!');
});

app.listen(3000, () => {
  console.log('Server running on port 3000');
});
